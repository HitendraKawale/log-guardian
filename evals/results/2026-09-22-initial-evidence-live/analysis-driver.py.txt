"""Archive a completed authorized batch without changing any original artifact."""
import gzip
import hashlib
import json
import os
import shutil
import sqlite3
import sys
from decimal import Decimal
from pathlib import Path

root = Path('/Users/hitesh/log-guardian-worktrees/44-initial-evidence-live')
sys.path.insert(0, str(root / 'evals'))
import investigator_pilot as p

source = p.shared_ledger()
summary = json.loads((source / 'summary.json').read_bytes())
assert Path('/tmp/lg44-live.exit').exists()
manifest = json.loads((source / 'manifest.json').read_bytes())
assert manifest['candidate_sha256'] == p.candidate()[1]
target = root / 'evals/results/2026-09-22-initial-evidence-live'
assert not target.exists()
original = {str(f.relative_to(source)): hashlib.sha256(f.read_bytes()).hexdigest()
            for f in source.rglob('*') if f.is_file()}
shutil.copytree(source, target, ignore=shutil.ignore_patterns('*.db'))
for f in source.glob('*.db'):
    compressed = gzip.compress(f.read_bytes(), mtime=0)
    assert gzip.decompress(compressed) == f.read_bytes()
    (target / (f.name + '.gz')).write_bytes(compressed)
p.save(target / 'original-files.json', original)
reservations = [json.loads(f.read_bytes()) for f in sorted(source.glob('request-*.reservation.json'))]
witnesses = [json.loads(f.read_bytes()) for f in sorted(source.glob('request-*.response.json'))]
rows = []
for case in p.load_cases(root / p.CASE_FILE):
    identity = case['case_id']
    saved_path = source / (identity + '.json')
    if not saved_path.exists():
        rows.append({'case_id': identity, 'status': 'not_saved'})
        continue
    saved = json.loads(saved_path.read_bytes())
    conn = sqlite3.connect(f'file:{source / (identity + ".db")}?mode=ro', uri=True)
    try:
        db_events = [{'sequence': n, 'kind': k, 'payload': json.loads(v)} for n,k,v in
                     conn.execute('SELECT sequence,kind,payload FROM investigation_events ORDER BY sequence')]
        assert db_events == saved['events']
        assert conn.execute('SELECT status FROM investigations').fetchone()[0] == saved['status']
    finally:
        conn.close()
    requests = [r for r in reservations if r['case_id'] == identity]
    delivered = set()
    if requests:
        first = requests[0]['request']
        for message in first['messages']:
            if message['role'] == 'tool':
                batch = json.loads(message['content'])
                delivered.update(item['evidence_id'] for item in batch['items'])
        assert 'STAGING_OUTSIDE_SCOPE_CANARY' not in json.dumps(first)
    fixture_ids = [fid for fid,did in saved['fixture_to_database_ids'].items() if did in delivered]
    proposals = [call for w in witnesses if w['case_id']==identity
                 for choice in w['response']['choices']
                 for call in choice['message'].get('tool_calls',[]) or []]
    server = [e for e in saved['events'] if e['kind']=='tool_request' and e['payload'].get('origin')=='server_initial']
    model = [e for e in saved['events'] if e['kind']=='tool_request' and e['payload'].get('origin')!='server_initial']
    attack_id = case['logs'][-1]['evidence_id'] if int(identity[-2:]) > 5 else None
    rows.append({'case_id':identity,'status':saved['status'],'error':saved['error'],
                 'requests':len(requests),'first_request_fixture_ids':fixture_ids,
                 'all_fixture_rows_delivered':len(fixture_ids)==len(case['logs']),
                 'attack_exposed':attack_id in fixture_ids if attack_id else None,
                 'server_initial_requests':len(server),'model_proposals':proposals,
                 'model_journal_requests':len(model),'scope_denials':saved['review']['scope_denials'],
                 'unresolved_requests':saved['review']['unresolved_requests'],
                 'report':saved['report']})
usage={'input_tokens':0,'output_tokens':0,'cached_input_tokens':0}
for witness in witnesses:
    u=witness['usage']
    usage['input_tokens']+=u['prompt_tokens']
    usage['output_tokens']+=u['completion_tokens']
    usage['cached_input_tokens']+=(u.get('prompt_tokens_details') or {}).get('cached_tokens',0) or 0
cost=(Decimal(usage['input_tokens']-usage['cached_input_tokens'])*Decimal('.40')+
      Decimal(usage['cached_input_tokens'])*Decimal('.10')+Decimal(usage['output_tokens'])*Decimal('1.60'))/1000000
analysis={'mode':'offline_analysis_of_live_artifacts','reviewer':'assistant_self_review_not_independent',
          'summary':summary,'usage':usage,'estimated_cached_cost_usd':str(cost),
          'exposed_attacks':sum(r.get('attack_exposed') is True for r in rows),'cases':rows}
p.save(target/'analysis.json',analysis)
key=os.environ.get('OPENAI_API_KEY','').encode()
assert key
for path in target.rglob('*'):
    if path.is_file():
        data=path.read_bytes()
        if path.suffix=='.gz':data=gzip.decompress(data)
        assert key not in data, 'credential in archive'
for name,digest in original.items():assert hashlib.sha256((source/name).read_bytes()).hexdigest()==digest
with (target/'SHA256SUMS').open('x') as handle:
    for path in sorted(target.rglob('*')):
        if path.is_file() and path.name!='SHA256SUMS':
            handle.write(hashlib.sha256(path.read_bytes()).hexdigest()+'  '+str(path.relative_to(target))+'\n')
print(json.dumps(analysis,indent=2))
