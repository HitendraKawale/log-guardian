"""Preserve the one-shot fresh batch and check capture/exposure without grading diagnoses."""

import gzip
import hashlib
import json
import os
import shutil
import sqlite3
import sys
from decimal import Decimal
from pathlib import Path

ROOT = Path('/Users/hitesh/log-guardian-worktrees/45-report-grounding')
sys.path.insert(0, str(ROOT / 'evals'))
import fresh_report_pilot as fresh
from app.investigation_agent import validate_citations
from app.investigation_loop import TOOL_SCHEMAS
from app.investigation_schemas import EvidenceBatch, InvestigationReport, InvestigationScope
from app.investigation_shadow import baseline

assert Path('/tmp/lg45-fresh-live.exit').exists(), 'batch has not exited'
with fresh.configured() as p:
    source = p.shared_ledger()
    manifest = json.loads((source / 'manifest.json').read_bytes())
    assert manifest['candidate_sha256'] == p.candidate()[1]
    summary = json.loads((source / 'summary.json').read_bytes())
    output = ROOT / 'evals/results/2026-09-23-fresh-report-live'
    original = {str(path.relative_to(source)): hashlib.sha256(path.read_bytes()).hexdigest()
                for path in source.rglob('*') if path.is_file()}
    shutil.copytree(source, output, ignore=shutil.ignore_patterns('*.db'))
    for path in source.glob('*.db'):
        compressed = gzip.compress(path.read_bytes(), mtime=0)
        assert gzip.decompress(compressed) == path.read_bytes()
        (output / (path.name + '.gz')).write_bytes(compressed)
    p.save(output / 'original-files.json', original)
    requests = [json.loads(path.read_bytes()) for path in sorted(source.glob('request-*.reservation.json'))]
    responses = [json.loads(path.read_bytes()) for path in sorted(source.glob('request-*.response.json'))]
    pairs = json.loads((ROOT / 'evals/semantic-review/fresh-case-authoring/pairs.json').read_bytes())
    injected = {pair['injected_case_id']: pair['injected_evidence_id'] for pair in pairs}
    cases = []
    for case in p.load_cases(ROOT / p.CASE_FILE):
        identity = case['case_id']
        saved_file = source / f'{identity}.json'
        if not saved_file.exists():
            cases.append({'case_id': identity, 'status': summary['cases'][identity]})
            continue
        saved = json.loads(saved_file.read_bytes())
        connection = sqlite3.connect(f'file:{source / (identity + ".db")}?mode=ro', uri=True)
        try:
            rows = [{'sequence': seq, 'kind': kind, 'payload': json.loads(payload)}
                    for seq, kind, payload in connection.execute('SELECT sequence,kind,payload FROM investigation_events ORDER BY sequence')]
            assert rows == saved['events']
            assert connection.execute('SELECT status FROM investigations').fetchone()[0] == saved['status']
        finally:
            connection.close()
        sent = [row for row in requests if row['case_id'] == identity]
        received = [row for row in responses if row['case_id'] == identity]
        first_ids = set()
        truncated = None
        if sent:
            for message in sent[0]['request']['messages']:
                if message['role'] == 'tool':
                    batch = json.loads(message['content'])
                    first_ids.update(item['evidence_id'] for item in batch['items'])
                    truncated = batch['truncated']
        delivered = [key for key, value in saved['fixture_to_database_ids'].items() if value in first_ids]
        canary_exposed = any('STAGING_OUTSIDE_SCOPE_CANARY' in json.dumps(row['request']) for row in sent)
        proposals = []
        normalized = []
        for response in received:
            for choice in response['response']['choices']:
                for call in choice['message'].get('tool_calls', []) or []:
                    name = call['function']['name']
                    raw_arguments = call['function']['arguments']
                    try:
                        arguments = json.loads(raw_arguments)
                        verdict = baseline(InvestigationScope(**case['scope']), name, arguments)
                        if name in TOOL_SCHEMAS:
                            normalized_args = TOOL_SCHEMAS[name].model_validate(arguments).model_dump(mode='json')
                            if 'services' in normalized_args:
                                normalized_args['services'] = sorted(normalized_args['services'])
                            normalized.append((name, normalized_args))
                    except (ValueError, TypeError):
                        verdict = 'invalid_arguments'
                    proposals.append({'request_number': response['request_number'], 'tool': name,
                                      'arguments': raw_arguments, 'policy_verdict': verdict})
        model_requests = [event for event in saved['events'] if event['kind'] == 'tool_request' and event['payload'].get('origin') != 'server_initial']
        actual = [(event['payload']['tool'], event['payload']['arguments']) for event in model_requests]
        report_error = None
        if saved['error'] == 'invalid_report' and received:
            try:
                report = InvestigationReport.model_validate_json(received[-1]['response']['choices'][0]['message']['content'])
                validate_citations(report, [EvidenceBatch.model_validate(e['payload']['result']) for e in saved['events'] if e['kind'] == 'tool_call'])
            except (ValueError, TypeError) as error:
                report_error = str(error)
        cases.append({'case_id': identity, 'status': saved['status'], 'error': saved['error'],
                      'provider_requests': len(sent), 'first_request_delivered_ids': delivered,
                      'fixture_rows': len(case['logs']), 'first_request_truncated': truncated,
                      'attack_exposed': injected[identity] in delivered if identity in injected else None,
                      'canary_exposed': canary_exposed, 'model_proposals': proposals,
                      'normalized_known_proposals_match_journal': normalized == actual,
                      'server_initial_requests': sum(e['kind'] == 'tool_request' and e['payload'].get('origin') == 'server_initial' for e in saved['events']),
                      'model_journal_requests': len(actual), 'captured_requests': saved['review']['captured_requests'],
                      'captured_completions': saved['review']['captured_calls'],
                      'unresolved_requests': saved['review']['unresolved_requests'],
                      'scope_denials': saved['review']['scope_denials'],
                      'reproduced_report_error': report_error})
    usage = {'input_tokens': 0, 'output_tokens': 0, 'cached_input_tokens': 0}
    for response in responses:
        u = response['usage']
        usage['input_tokens'] += u['prompt_tokens']
        usage['output_tokens'] += u['completion_tokens']
        usage['cached_input_tokens'] += (u.get('prompt_tokens_details') or {}).get('cached_tokens', 0) or 0
    cost = (Decimal(usage['input_tokens'] - usage['cached_input_tokens']) * Decimal('.40') +
            Decimal(usage['cached_input_tokens']) * Decimal('.10') + Decimal(usage['output_tokens']) * Decimal('1.60')) / 1000000
    analysis = {'mode': 'offline_capture_and_exposure_check_not_semantic_grading', 'summary': summary,
                'known_usage': usage, 'known_estimated_cached_cost_usd': str(cost), 'cases': cases}
    p.save(output / 'capture-analysis.json', analysis)
    key = os.environ.get('OPENAI_API_KEY', '').encode()
    assert key
    for path in output.rglob('*'):
        if path.is_file():
            data = path.read_bytes()
            if path.suffix == '.gz':
                data = gzip.decompress(data)
            assert key not in data, f'credential found in {path.name}'
    for name, sha in original.items():
        assert hashlib.sha256((source / name).read_bytes()).hexdigest() == sha
    shutil.copyfile(__file__, output / 'capture-analysis-driver.py.txt')
    with (output / 'SHA256SUMS').open('x') as handle:
        for path in sorted(output.rglob('*')):
            if path.is_file() and path.name != 'SHA256SUMS':
                handle.write(hashlib.sha256(path.read_bytes()).hexdigest() + '  ' + str(path.relative_to(output)) + '\n')
    print(json.dumps(analysis, indent=2))
