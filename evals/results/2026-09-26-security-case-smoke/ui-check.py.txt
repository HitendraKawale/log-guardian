import json
from pathlib import Path
from playwright.sync_api import sync_playwright, expect

state=json.loads(Path('/tmp/lg48-live-viewer-runtime.json').read_text())
out=Path('/tmp/lg48-live-ui');out.mkdir(exist_ok=True)
with sync_playwright() as p:
    browser=p.chromium.launch(headless=True)
    for n in range(1,5):
        base=f'http://smoke-{n:02}.localhost:8484'
        page=browser.new_page(viewport={'width':1280,'height':1000})
        errors=[];page.on('pageerror',lambda error: errors.append(str(error)))
        page.goto(base+'/ui/security.html?api='+base)
        page.get_by_label('Security API key',exact=True).fill(state['review_key'])
        page.get_by_role('button',name='Connect',exact=True).click()
        expect(page.locator('#security-history button')).to_have_count(1)
        page.locator('#security-history button').click()
        expect(page.locator('#security-detail')).to_be_visible()
        page.get_by_label('Investigation API key',exact=True).fill(state['execution_key'])
        page.get_by_label('I authorize sharing this case with the model provider.',exact=True).check()
        page.get_by_role('button',name='Start or open investigation',exact=True).click()
        expected='completed' if n==3 else 'failed: invalid_report'
        expect(page.locator('#security-run-status')).to_contain_text(expected)
        if n==3:
            expect(page.locator('#security-run-report')).to_contain_text('Inconclusive',ignore_case=True)
            page.locator('#security-run-report .citation').first.click()
            expect(page.locator('#security-run-evidence details[open] pre')).to_contain_text('http_status')
        else:
            expect(page.locator('#security-run-report')).to_be_empty()
        assert page.request.post(base+'/investigations',headers={'X-API-Key':state['execution_key']},data='{}').status==405
        assert page.evaluate('JSON.stringify(localStorage)+JSON.stringify(sessionStorage)')=='{}{}'
        assert not errors
        page.screenshot(path=str(out/f'smoke-{n:02}.png'),full_page=True)
        print(f'smoke-{n:02}: {expected}; read-only POST 405; no browser errors')
        page.close()
    browser.close()
