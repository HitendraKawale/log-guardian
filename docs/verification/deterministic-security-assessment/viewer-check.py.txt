"""Check no-model assessments against copies of the closed live batch."""
import json
from pathlib import Path
from playwright.sync_api import sync_playwright, expect
state=json.loads(Path('/tmp/lg48-deterministic-viewer-runtime.json').read_text())
out=Path('/tmp/lg48-deterministic-viewer-proof');out.mkdir(exist_ok=True)
with sync_playwright() as p:
    browser=p.chromium.launch(headless=True)
    for n,count in enumerate([19,8,6,8],1):
        url=f'http://smoke-{n:02}.localhost:8488'
        page=browser.new_page(viewport={'width':1280,'height':1000})
        errors=[];page.on('pageerror',lambda error:errors.append(str(error)))
        page.goto(url+'/ui/security.html?api='+url)
        page.get_by_label('Security API key',exact=True).fill(state['review_key'])
        page.get_by_role('button',name='Connect',exact=True).click()
        expect(page.locator('#security-history button')).to_have_count(1)
        page.locator('#security-history button').click()
        expect(page.locator('#security-assessment-status')).to_contain_text(f'Reviewed {count} of {count}')
        expect(page.locator('#security-assessment-status')).to_contain_text('No model call')
        expect(page.get_by_label('Investigation API key',exact=True)).to_have_value('')
        page.locator('#security-assessment-report .citation').filter(has_text='security-event:').first.click()
        expect(page.locator('#security-assessment-evidence details[open] pre')).to_be_visible()
        runs=page.request.get(url+'/investigations',headers={'X-API-Key':state['execution_key']}).json()
        assert len(runs)==1
        expected='completed' if n in (1,3) else 'failed'
        assert runs[0]['status']==expected
        if n in (2,4):assert runs[0]['error']=='invalid_report' and runs[0]['report'] is None
        if n==3:
            expect(page.locator('#security-assessment-report')).to_contain_text('Authentication outcome is unknown')
            expect(page.locator('#security-assessment-report')).not_to_contain_text('Authentication service recorded success')
        assert page.request.post(url+'/investigations',data={}).status==405
        assert page.evaluate('JSON.stringify(localStorage)+JSON.stringify(sessionStorage)')=='{}{}'
        page.screenshot(path=str(out/f'smoke-{n:02}-desktop.png'),full_page=True)
        page.set_viewport_size({'width':390,'height':844})
        assert page.evaluate('document.documentElement.scrollWidth <= innerWidth')
        page.screenshot(path=str(out/f'smoke-{n:02}-mobile.png'),full_page=True)
        assert not errors
        print(f'smoke-{n:02}: baseline reviewed {count}/{count}; historical AI {expected}; citations opened; POST 405; storage empty; desktop/mobile no overflow or page errors')
        page.close()
    browser.close()
print('PASS: 41 saved records reviewed without model execution; historical reports unchanged.')
